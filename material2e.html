  • B - Usar o evento "keyup" em vez de "submit"

Explicação do JS

Código utilizado:

const campoNome = document.getElementById('nome');
            const campoEmail = document.getElementById('email');
            const erroNome = document.getElementById('erroNome');
            const erroEmail = document.getElementById('erroEmail');
            
            campoNome.addEventListener('keyup', function() {
                const nome = campoNome.value.trim();
            
                if (nome.length < 3) {
                    erroNome.textContent = 'O nome deve ter pelo menos 3 letras.';
                    campoNome.className = 'erro';
                } else {
                    erroNome.textContent = '';
                    campoNome.className = 'ok';
                }
            });
            
            campoEmail.addEventListener('keyup', function() {
                const email = campoEmail.value.trim();
            
                if (!email.includes('@')) {
                    erroEmail.textContent = 'Digite um e-mail válido com @.';
                    campoEmail.className = 'erro';
                } else {
                    erroEmail.textContent = '';
                    campoEmail.className = 'ok';
                }
            });

Traduzindo o sucesso do nome: "SENÃO (quando atingir 3 ou mais letras), apague a mensagem de erro e aplique a classe 'ok' (borda verde) no campo."
